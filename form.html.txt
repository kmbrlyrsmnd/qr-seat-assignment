<!DOCTYPE html>
<html>
<head>
  <title>Enter Name</title>
</head>
<body>
  <h2>Enter your name</h2>
  <form id="nameForm">
    <input type="text" id="nameInput" required>
    <button type="submit">Get Seat Number</button>
  </form>
  <p id="seatResult"></p>

  <script>
    const seatMap = {
      "Alice": "A1",
      "Bob": "B2",
      "Charlie": "C3"
    };

    document.getElementById("nameForm").onsubmit = function(e) {
      e.preventDefault();
      const name = document.getElementById("nameInput").value.trim();
      const seat = seatMap[name] || "Seat not found";
      document.getElementById("seatResult").textContent = `Your seat: ${seat}`;
    };
  </script>
</body>
</html>
